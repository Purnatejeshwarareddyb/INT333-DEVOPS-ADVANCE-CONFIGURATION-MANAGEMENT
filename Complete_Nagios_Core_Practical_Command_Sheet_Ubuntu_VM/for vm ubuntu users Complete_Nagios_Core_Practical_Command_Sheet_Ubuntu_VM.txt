
The Nginx configuration included:

```nginx
location /nagios/ {
    alias /usr/local/nagios/share/;
    index index.php;

    auth_basic "Nagios";
    auth_basic_user_file /etc/nginx/.htpasswd;
}
```

PHP:

```nginx
location ~ ^/nagios/(.+\.php)$ {
    alias /usr/local/nagios/share/$1;

    include fastcgi_params;
    fastcgi_param SCRIPT_FILENAME /usr/local/nagios/share/$1;
    fastcgi_param SCRIPT_NAME /nagios/$1;

    fastcgi_pass unix:/run/php/php8.5-fpm.sock;
}
```

CGI:

```nginx
location ~ ^/nagios/cgi-bin/(.+\.cgi)$ {
    include fastcgi_params;

    fastcgi_param SCRIPT_FILENAME /usr/local/nagios/sbin/$1;
    fastcgi_param SCRIPT_NAME /nagios/cgi-bin/$1;
    fastcgi_param DOCUMENT_ROOT /usr/local/nagios/sbin;

    fastcgi_pass unix:/run/fcgiwrap.socket;
}
```
